### Implementation Concept

```ts
function primMST(n: number, adjList: {node: number, weight: number}[][]): number {
  const visited = new Set<number>();
  const pq = new MinPriorityQueue((a, b) => a[0] - b[0]);
  
  // Start arbitrarily at node 0
  pq.push([0, 0]); // [edgeCost, node]
  let mstCost = 0;

  while (!pq.isEmpty() && visited.size < n) {
    const [cost, u] = pq.pop();

    if (visited.has(u)) continue; // Already in the MST
    
    // Add to MST
    visited.add(u);
    mstCost += cost;

    // Evaluate neighbors
    for (const { node: v, weight } of adjList[u]) {
      if (!visited.has(v)) {
        pq.push([weight, v]); // Note: We push ONLY the edge weight, not cost + weight
      }
    }
  }

  return visited.size === n ? mstCost : -1;
}
```
